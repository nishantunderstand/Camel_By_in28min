package camel_microservice_a.route.a;

import org.apache.camel.builder.RouteBuilder;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.time.LocalDateTime;

@Autowired
private GetCurrentTimeBean getCurrentTimeBeanAutowired;

@Component
public class MyFirstRouter2026 extends RouteBuilder {
    @Override
    public void configure() throws Exception {
        from("timer:first-timer")
                //.transform().constant("My Constant Message")
                //.transform().constant("Time Now is "+ LocalDateTime.now())
                //.bean("getCurrentTimeBean")
                .bean(getCurrentTimeBeanAutowired)
                .to("log:first-timer");
    }
}



@Component
class GetCurrentTimeBean{
    public String getCurrentTime(){
        return "Time Now are => "+ LocalDateTime.now();
    }
}
